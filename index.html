document.addEventListener('DOMContentLoaded', function() { const cards = document.querySelectorAll('.card'); const footerBox = document.querySelector('.footer-box'); // Tüm kartlara ve footer'a visible sınıfını ekleyen fonksiyon function makeVisible() { cards.forEach(card => { card.classList.add('visible'); }); if (footerBox) { footerBox.classList.add('visible'); } } // Eğer IntersectionObserver destekleniyorsa, onu kullan if ('IntersectionObserver' in window) { const observer = new IntersectionObserver((entries) => { entries.forEach(entry => { if (entry.isIntersecting) { entry.target.classList.add('visible'); } }); }, { threshold: 0.1 }); cards.forEach(card => { observer.observe(card); }); if (footerBox) { observer.observe(footerBox); } } else { // IntersectionObserver desteklenmiyorsa, hemen göster makeVisible(); } });
